## 10. แบบจำลองข้อมูล

### 10.1 หลักการ

- ทุกตารางมี `id` เป็น `cuid2`, `createdAt`, `updatedAt`
- ตารางที่ผู้ใช้สร้างเนื้อหามี `deletedAt` (soft delete) และ query ต้องกรองออกเสมอ
- เงินทุกช่องเป็น `Int` หน่วยสตางค์ ชื่อลงท้าย `Satang`
- ห้ามใช้ `String` แทน enum ที่รู้ค่าแน่นอน
- ทุก foreign key ต้องมี index ถ้าใช้ในเงื่อนไขค้นหา
- ตารางที่โตเร็ว (`ListingView`, `Notification`, `OutboxEvent`) ต้องมีแผน partition/ลบเก่า

### 10.3 SQL เสริมที่ต้องรันหลัง migrate

```sql
-- ค้นหาภาษาไทยด้วย trigram
CREATE EXTENSION IF NOT EXISTS pg_trgm;
CREATE EXTENSION IF NOT EXISTS unaccent;

CREATE INDEX listing_title_trgm ON "Listing" USING gin (title gin_trgm_ops);
CREATE INDEX shop_name_trgm ON "Shop" USING gin (name gin_trgm_ops);

-- tsvector สำหรับค้นหาสำรองเมื่อ Meilisearch ล่ม
ALTER TABLE "Listing" ADD COLUMN IF NOT EXISTS "searchVector" tsvector
  GENERATED ALWAYS AS (
    setweight(to_tsvector('simple', coalesce(title,'')), 'A') ||
    setweight(to_tsvector('simple', coalesce(description,'')), 'B')
  ) STORED;
CREATE INDEX listing_search_idx ON "Listing" USING gin ("searchVector");

-- ค้นหาตามระยะทาง
CREATE EXTENSION IF NOT EXISTS postgis;
ALTER TABLE "Listing" ADD COLUMN IF NOT EXISTS geom geography(Point, 4326);
CREATE INDEX listing_geom_idx ON "Listing" USING gist (geom);

-- กันสั่งซื้อเกินสต็อก (ป้องกันในระดับฐานข้อมูล)
ALTER TABLE "Variant" ADD CONSTRAINT variant_qty_nonneg CHECK (quantity >= 0);
ALTER TABLE "Listing" ADD CONSTRAINT listing_price_positive CHECK ("priceSatang" > 0);
ALTER TABLE "Wallet" ADD CONSTRAINT wallet_nonneg
  CHECK ("availableSatang" >= 0 AND "pendingSatang" >= 0 AND "reservedSatang" >= 0);
```
